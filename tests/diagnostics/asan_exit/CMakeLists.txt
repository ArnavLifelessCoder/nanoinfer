cmake_minimum_required(VERSION 3.20)
project(asan_exit_probe LANGUAGES CXX)
if(NOT MSVC)
  message(FATAL_ERROR "This reproducer targets the MSVC AddressSanitizer runtime")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDebugDLL)
option(PROBE_ASAN "Instrument the isolated probe with MSVC ASan" ON)
option(PROBE_FLUSH "Explicitly flush stdout before returning" OFF)
set(CMAKE_CXX_FLAGS_DEBUG "/O1 /Zi")
add_executable(asan_exit_probe probe.cpp)
target_compile_options(asan_exit_probe PRIVATE /W4 /WX /permissive-)
if(PROBE_ASAN)
  target_compile_options(asan_exit_probe PRIVATE /fsanitize=address)
endif()
if(PROBE_FLUSH)
  target_compile_definitions(asan_exit_probe PRIVATE PROBE_FLUSH)
endif()
target_link_options(asan_exit_probe PRIVATE /INCREMENTAL:NO /DEBUG)
add_executable(asan_error_probe error_probe.cpp)
target_compile_options(asan_error_probe PRIVATE /W4 /WX /permissive- /fsanitize=address)
target_link_options(asan_error_probe PRIVATE /INCREMENTAL:NO /DEBUG)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME asan_exit_repro
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/repro.py"
    --executable "$<TARGET_FILE:asan_exit_probe>"
    --directory "${CMAKE_CURRENT_BINARY_DIR}/evidence"
    --procdump "${PROCDUMP_EXECUTABLE}")
